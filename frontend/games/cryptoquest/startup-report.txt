  <script>
    (async()=>{
      try{
        const names=Array.from({length:12},(_,i)=>`data/p${String(i).padStart(2,'0')}.txt`);
        const parts=await Promise.all(names.map(async name=>{
          const response=await fetch(name,{cache:'no-store'});
          if(!response.ok)throw new Error(`${name}: ${response.status}`);
          return response.text();
        }));
        const packed=Uint8Array.from(atob(parts.join('')),char=>char.charCodeAt(0));
        const stream=new DecompressionStream('gzip');
        let html=await new Response(new Blob([packed]).stream().pipeThrough(stream)).text();
        const visualHead='<link rel="stylesheet" href="/games/cryptoquest/v31-god-mode.css?v=31.1.0"><meta name="cryptoquest-presentation" content="V31.1-GOD-MODE">';
        const visualRuntime='<script src="/games/cryptoquest/v31-god-mode-runtime.js?v=31.1.0" defer><\/script>';
        html=html.includes('</head>')?html.replace('</head>',visualHead+'</head>'):visualHead+html;
        html=html.includes('</body>')?html.replace('</body>',visualRuntime+'</body>'):html+visualRuntime;
        document.open();
        document.write(html);
        document.close();
      }catch(error){
        document.getElementById('err').textContent='No se pudo cargar CryptoQuest: '+error.message;
      }
    })();
  </script>
</body>
