function resolveZoneEventChoice(choice){
  const pending=game.mapEvent?.pendingFight,event=ZONE_EVENTS.find(x=>x.id===game.mapEvent?.eventId);if(!pending||!event)return;
  const {mi,ni}=pending,map=MAPS[mi],combatChoices=['fight','stabilize','enter','defend','open','purify'];game.mapEvent.pending=false;addZoneMastery(game,map.id,choice==='ignore'?1:4,'evento');
  if(choice==='absorb'){game.temporaryBuffs.push({id:`corrupt-${Date.now()}`,name:'Poder corrupto',stats:{attack:8,magic:8,crit:4},expiresCombats:3,consequence:'curse'});game.factions.aether.value=Math.max(0,game.factions.aether.value-10);toast('Poder absorbido · 3 combates · reputación −10','success');}
  else if(choice==='pray'){restAtCity(game);toast('Descansas en el santuario; HP y maná se recuperan con el tiempo','success');}
  else if(['escort','search'].includes(choice)){deliverRewards(game,{gold:choice==='escort'?90:35,materials:{iron:2,crystal:1},reputation:{iron:choice==='escort'?25:0}},{transactionId:`event:${event.id}:${Date.now()}`,save:false});toast('Recompensa del evento recibida','success');}
  else if(choice==='ignore')toast('Continúas tu camino');
  if(event.kind==='merchant'){game.mapEvent=null;saveGame(game);game.activeTab='more';game.moreScreen='merchant';render();return;}
  game.mapEvent=null;saveGame(game);if(combatChoices.includes(choice)||(event.kind==='combat'&&choice!=='ignore'))startFightV9(mi,ni,{skipEvent:true,forceElite:event.elite||['stabilize','defend','open','purify'].includes(choice),forceRare:event.rare||choice==='enter',eventChoice:event.id});else startFightV9(mi,ni,{skipEvent:true});
}
